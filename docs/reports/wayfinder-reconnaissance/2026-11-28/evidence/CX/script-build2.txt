def weighted(sample,per,strat,keyl,keye,stratsize):
    num=den=0
    for c in CL:
        s=[r for r in sample if r['period']==per and r[strat]==c]
        if not s: continue
        w=stratsize[c]/len(s)
        num+=w*sum(float(r[keye]) for r in s); den+=w*sum(float(r[keyl]) for r in s)
    return 1000*num/den
def unweighted(sample,per,keyl,keye):
    s=[r for r in sample if r['period']==per]
    return 1000*sum(float(r[keye]) for r in s)/sum(float(r[keyl]) for r in s)
