"""Declared project-to-project dependencies from Gradle build files (static text parse).
usage: gradle_declared.py <repo> <out.csv> module=builddir ...
Matches `<configuration>(...project(":x")...)` and `"<configuration>"(project(":x"))` in each module's
build.gradle(.kts). Convention plugins under build-logic are scanned and reported with module '(convention:<file>)'.
"""
import sys, os, re, csv, glob
repo, out = sys.argv[1:3]
rows=[]
R = re.compile(r'^\s*"?([A-Za-z]+)"?\s*\(?\s*(testFixtures\()?\s*(platform\()?\s*project\(\s*"(:[\w-]+)"\s*\)(.*)$')
def scan(mod, path):
    if not os.path.exists(path): return
    for i,l in enumerate(open(path,encoding="utf-8"),1):
        if l.lstrip().startswith("//"): continue
        m = R.match(l)
        if m: rows.append([mod, m.group(1), m.group(4).lstrip(":") + ("[testFixtures]" if m.group(2) else ""), os.path.relpath(path,repo), i])
        elif "project(\"" in l and "projectDir" not in l:
            rows.append([mod, "(other-reference)", re.search(r'project\(\s*"(:[\w$-]+)"', l).group(1).lstrip(":") if re.search(r'project\(\s*"(:[\w$-]+)"', l) else l.strip()[:60], os.path.relpath(path,repo), i])
for a in sys.argv[3:]:
    mod, d = a.split("=",1)
    for fn in ("build.gradle.kts","build.gradle"): scan(mod, os.path.join(repo,d,fn))
for p in sorted(glob.glob(os.path.join(repo,"build-logic/src/main/kotlin/*.kts"))): scan("(convention:%s)"%os.path.basename(p), p)
with open(out,"w",newline="") as fh:
    w=csv.writer(fh); w.writerow(["module","configuration","target","file","line"]); w.writerows(rows)
for r in rows: print(",".join(map(str,r)))
