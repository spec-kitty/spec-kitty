cn = sorted(nodes)


def closure(edge_keys):
    adj = defaultdict(set)
    for a, b in edge_keys:
        adj[a].add(b)
    out = {}
    for n in cn:
        seen = set()
        st = [n]
        while st:
            x = st.pop()
            for y in adj[x]:
                if y not in seen:
                    seen.add(y)
                    st.append(y)
        out[n] = seen
    return out


cl_all = closure(cw)
cl_load = closure(clw)
indeg = Counter(b for a, b in cw)
outdeg = Counter(a for a, b in cw)

mutual = sorted({tuple(sorted((a, b))) for (a, b) in cw if (b, a) in cw})

    "component_sccs_all": [sorted(c) for c in scc_all],
    "component_sccs_load_time": [sorted(c) for c in scc_load],

        "possible_directed_pairs": len(cn) * (len(cn) - 1),
        "components_in_cycle_all": len(in_all),

        "components_in_cycle_load_time": len(in_load),
        "largest_component_scc_load_time": max(map(len, scc_load), default=0),

        "module_edges_cross_component_load_time": sum(clw.values()),

}

    """Module strings the gate's collector would see for one import record (resolved form)."""
    if r["kind"] == "from" and r.get("name_is_module"):
        return [r["target"], r["resolved"]]
    return [r["target"]]

# ---------------------------------------------------------------- G. external contract packages
EXT = {"spec_kitty_events", "spec_kitty_tracker"}
ext_all: dict[str, list[str] | None] = {}


def ext_module_all(modname: str):
    if site is None:
        return None
    if modname in ext_all:
        return ext_all[modname]
    p = site.joinpath(*modname.split("."))
    f = p / "__init__.py" if p.is_dir() else p.with_suffix(".py")
    res = None
    if f.is_file():
        tree = ast.parse(f.read_text())
        for n in tree.body:
            tgt = None
            if isinstance(n, ast.Assign) and any(isinstance(t, ast.Name) and t.id == "__all__" for t in n.targets):
                tgt = n.value
            elif isinstance(n, ast.AnnAssign) and isinstance(n.target, ast.Name) and n.target.id == "__all__":
                tgt = n.value
            if tgt is not None and isinstance(tgt, (ast.List, ast.Tuple)):
                res = [e.value for e in tgt.elts if isinstance(e, ast.Constant)]
    else:
        res = "MODULE_NOT_FOUND"
    ext_all[modname] = res
    return res
