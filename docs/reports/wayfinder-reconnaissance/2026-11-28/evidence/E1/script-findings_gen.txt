def sha(f):
    return hashlib.sha256(open(EV + f, 'rb').read()).hexdigest()


def ev(i, kind, f, method, exit_code=0):
    return dict(id=f'E{i}', kind=kind, reference='evidence/E1/' + f, method=method, sha256=sha(f), exit_code=exit_code, environment=ENV, observed_at=AT)


def doc(i, ref, what):
    return dict(id=f'E{i}', kind='document-claim', reference=ref, method=what, sha256=None, exit_code=None, environment='read at the named revision', observed_at=AT)

def F(measure, component, repo, rev, claim, evidence, covered, not_covered, dis_search, dis_result, *, cls='observation', view='observed', kind='head', conf='high', pop=None, sample=None, seed=None, deviation=None, counter=None, interp=None, derived=None, rel=None, qa=None, inv=None, oq=None):
    N[0] += 1
    fid = f'F-E1-{N[0]:03d}'
    d = dict(
        finding_id=fid, workstream='E1 - Assurance inventory, exceptions, and the research-mission experiment', hypotheses=[],
        classification=cls, view=view, prereg=dict(measure=measure, deviation=deviation),
        subject=dict(component=component, repository='spec-kitty/' + repo, revision=rev, revision_kind=kind),
        claim=claim, evidence=evidence,
        scope=dict(covered=covered, not_covered=not_covered, population=pop, sample=sample, seed=seed),
        interpretation=interp, derived_from=derived or [], counterevidence=counter or [],
        disconfirmation=dict(search=dis_search, result=dis_result), relations=rel or [], confidence=conf,
        analyst=dict(profile='reviewer-renata', model='fable'), verification=dict(status='unverified', by=None),
        quality_attributes=qa or [], candidate_invariants=inv or [], open_questions=oq or [])
    ALL.append(d)
    yaml.safe_dump(d, open(OUT + fid + '.yaml', 'w'), sort_keys=False, allow_unicode=True, width=110)
    return fid
