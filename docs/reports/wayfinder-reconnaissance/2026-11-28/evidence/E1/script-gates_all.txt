def cat(name):
    n = name.lower()
    for c, p in RULES:
        if re.search(p, n):
            return c
    return 'other'


def mech(src):
    m = []
    if 'pytestarch' in src or 'LayerRule' in src:
        m.append('pytestarch layer rule')
    if re.search(r'\bast\.(parse|walk|NodeVisitor)|_ast_scan|_scan import', src):
        m.append('AST scan')
    if re.search(r'"ls-files"|"grep"|git grep|git ls-files', src):
        m.append('git grep / ls-files')
    if re.search(r'yaml\.safe_load|json\.load', src):
        m.append('committed data file compared with live tree')
    if re.search(r're\.(compile|search|findall|match)', src) and 'AST scan' not in m:
        m.append('regex text scan')
    if re.search(r'subprocess|CliRunner|runner\.invoke', src):
        m.append('runs a subprocess or the CLI')
    return '; '.join(m) or 'pytest assertions over imported code'
