def ylist(pathexpr,idf,ctx=None):
    def f(t):
        try: d=yaml.safe_load(t)
        except Exception: return None
        cur=d
        for k in pathexpr:
            if not isinstance(cur,dict) or k not in cur: return {}
            cur=cur[k]
        out={}
        for e in (cur or []):
            if isinstance(e,dict):
                key=J({k:e.get(k) for k in idf}) if idf else J(e)
                blob=J(e)+(' || '+ctx(d,e) if ctx else '')
            else: key=J(e); blob=J(e)
            out[key]=blob
        return out
    return f

def txtlines(t):
    out={}; block=[]
    for l in t.splitlines():
        s=l.strip()
        if not s: block=[]; continue
        if s.startswith('#'): block.append(s); continue
        out[s]=' '.join(block[-12:])+' || '+s
    return out
def pydict_lines(symbol):
    def f(t):
        m=re.search(r'^%s[^\n]*=[^\n]*[{(]\s*$'%re.escape(symbol),t,re.M)
        if not m: return {}
        out={}; block=[]
        for l in t[m.end():].splitlines():
            s=l.strip()
            if s.startswith('}') or s.startswith(')'): break
            if s.startswith('#'): block.append(s); continue
            mm=re.match(r'''["']([^"']+)["']\s*(?::\s*["']([^"']+)["'])?''',s)
            if mm: out[mm.group(1)]=' '.join(block)+' || '+s; 
            if not s.startswith('#') and mm: block=[]
        return out
    return f
def toml_table(section_re):
    def f(t):
        m=re.search(section_re,t,re.M)
        if not m: return {}
        out={}; block=[]
        for l in t[m.end():].splitlines():
            s=l.strip()
            if s.startswith('['): break
            if s.startswith('#'): block.append(s); continue
            mm=re.match(r'''["']([^"']+)["']\s*=\s*(.*)''',s)
            if mm: out[mm.group(1)]=' '.join(block[-4:])+' || '+s
            if not s: block=[]
        return out
    return f
def toml_array(start_re):
    def f(t):
        m=re.search(start_re,t,re.M)
        if not m: return {}
        out={}; block=[]
        for l in t[m.end():].splitlines():
            s=l.strip()
            if s.startswith(']'): break
            if s.startswith('#'): block.append(s); continue
            for q in re.findall(r'''["']([^"']+)["']''',s.split('#')[0]): out[q]=' '.join(block[-4:])+' || '+s
        return out
    return f
def baselines_leaves(t):
    out={}; sec=None; block=[]
    for l in t.splitlines():
        if not l.strip(): continue
        if l.strip().startswith('#'): block.append(l.strip()); continue
        m=re.match(r'^(\w+):\s*$',l)
        if m: sec=m.group(1); continue
        m=re.match(r'^\s+(\w+):\s*(\d+)\s*(#.*)?$',l)
        if m and sec: out[f'{sec}.{m.group(1)}']=(int(m.group(2)),' '.join(block[-8:])+' '+(m.group(3) or '')); block=[]
    return out
