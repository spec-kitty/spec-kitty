import json,sys,re,fnmatch,subprocess,random,csv,collections
S=sys.argv[1]; out=json.load(open(S+'/own_commits.json'))
REQ=re.compile(r'^[+-]\s*"?[A-Za-z0-9_.-]+(\[[^\]]*\])?\s*(==|>=|<=|~=|<|>)')
def pin(paths,sha):
    for p in paths:
        if fnmatch.fnmatch(p,'packs/*.source.yaml') or p.startswith('pins/') or p=='uv.lock': return True
        if p=='pyproject.toml':
            d=subprocess.run(['git','-C','/home/user/spec-kitty','show','--format=','-U0',sha,'--','pyproject.toml'],capture_output=True,text=True).stdout
            if any(REQ.match(l) for l in d.splitlines() if not l.startswith(('+++','---'))): return True
    return False
def contract(paths):
    for p in paths:
        if p.startswith(('kitty-specs/','tests/')): continue
        segs=p.split('/'); b=segs[-1]
        if any(s in('contracts','contract','schemas','schema') for s in segs[:-1]) or fnmatch.fnmatch(b,'*.schema.json') or fnmatch.fnmatch(b,'*schema*.yaml') or fnmatch.fnmatch(b,'*schema*.yml') or fnmatch.fnmatch(b,'openapi*'): return True
    return False
A=[];B=[];nei=0;win=0
for o in out:
    if not(o['line']=='converged' and '2026-09-07'<=o['day']<='2026-11-28' and o['fix'] and not o['bulk']): continue
    win+=1
    if o['n']>=2 or pin(o['paths'],o['sha']) or contract(o['paths']): A.append(o['sha'])
    elif o['n']==1: B.append(o['sha'])
    else: nei+=1
print('window fix commits',win,'A',len(A),'B',len(B),'neither',nei)
